# ZeoPlant FR/DE/AR i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add French, German and Arabic (RTL) to the single-file ZeoPlant PWA with a header switcher, remembered choice and per-language number formatting.

**Architecture:** One `I18N` dictionary + `t()`/`setLang()`/`applyLang()` in the existing inline script; static text tagged with `data-i18n`, dynamic text in `calc()` routed through `t()`. CSS physical left/right rules become logical so `dir="rtl"` mirrors the page. Verification is a Playwright script kept in the session scratchpad (not shipped).

**Tech Stack:** plain HTML/CSS/JS, `Intl.NumberFormat`, Playwright (installed in the scratchpad of the PWA build session) for verification.

## Global Constraints
- Single runtime file: all code lives in `index.html`; no new files served by `sw.js`.
- Brand "ZeoPlant", currency "DH"/"MAD" and units (m², m³, kg, kg/m³, L/m²·d, L/day, %) stay Latin in every language.
- Arabic numbers use Western digits (`ar-MA` + `numberingSystem:'latn'`).
- Locales: en→`en-US`, fr→`fr-FR`, de→`de-DE`, ar→`ar-MA`.
- Stored keys: inputs `zeoplant.inputs.v1` (existing), language `zeoplant.lang` (new).
- `../ZeoPlant-Dashboard.html` and `manifest.webmanifest` are not modified.
- After the change bump `sw.js` CACHE to `zeoplant-v2`.

**Files:**

**Interfaces:**

- [ ] **Step 1: Write the script**

```js
import { chromium } from 'playwright';
import { pathToFileURL } from 'url';
import path from 'path';
const file = pathToFileURL(process.argv[2]).href;
const outDir = path.dirname(new URL(import.meta.url).pathname.replace(/^\/([A-Z]:)/, '$1'));
const EXPECT = { en:{dir:'ltr', n:'1,234.5'}, fr:{dir:'ltr', n:'1 234,5'}, de:{dir:'ltr', n:'1.234,5'}, ar:{dir:'rtl', n:'1,234.5'} };
const fails = [];
const browser = await chromium.launch();
const page = await browser.newPage({ viewport:{ width:390, height:844 } });
await page.goto(file);
const keyReport = await page.evaluate(() => {
  const ref = Object.keys(I18N.en).sort();
  return Object.fromEntries(Object.keys(I18N).map(l => {
    const ks = Object.keys(I18N[l]);
    return [l, { missing: ref.filter(k => !ks.includes(k)), extra: ks.filter(k => !ref.includes(k)) }];
  }));
});
for (const [l, r] of Object.entries(keyReport)) if (r.missing.length || r.extra.length) fails.push(`${l}: missing ${r.missing} extra ${r.extra}`);
const enTexts = {};
for (const lang of ['en','fr','de','ar']) {
  await page.click(`.brand-meta button[data-lang="${lang}"]`, { force: true });
  const html = await page.evaluate(() => [document.documentElement.lang, document.documentElement.dir]);
  if (html[0] !== lang || html[1] !== EXPECT[lang].dir) fails.push(`${lang}: html lang/dir = ${html}`);
  const num = await page.evaluate(() => fmt(1234.5, 1));
  if (num.replace(/ | /g,' ') !== EXPECT[lang].n) fails.push(`${lang}: fmt(1234.5)=${JSON.stringify(num)}`);
  const texts = await page.evaluate(() => [...document.querySelectorAll('[data-i18n],[data-i18n-html]')].map(e => e.textContent.trim()));
  if (lang === 'en') texts.forEach((t,i) => enTexts[i] = t);
  else {
    const same = texts.filter((t,i) => t === enTexts[i] && t.length > 3 && !/^(ZeoPlant|ROI|DH|Σ)/.test(t));
    if (same.length > 6) fails.push(`${lang}: ${same.length} nodes still English: ${same.slice(0,8).join(' | ')}`);
  }
  const raw = await page.evaluate(() => (document.body.innerText.match(/\b[a-z]+\.[a-z0-9]+\b/g) || []).filter(k => !k.endsWith('.xlsx')));
  if (raw.length) fails.push(`${lang}: raw keys visible: ${raw.join(',')}`);
  const stored = await page.evaluate(() => localStorage.getItem('zeoplant.lang'));
  if (stored !== lang) fails.push(`${lang}: not persisted (${stored})`);
  await page.screenshot({ path: path.join(outDir, `shot-${lang}.png`), fullPage: true });
}
await page.reload();
const after = await page.evaluate(() => document.documentElement.lang);
if (after !== 'ar') fails.push(`reload lost language: ${after}`);
await browser.close();
if (fails.length) { console.error('FAIL\n' + fails.join('\n')); process.exit(1); }
console.log('PASS: 4 languages, dictionaries complete, RTL, numbers, persistence');
```

### Task 3: Dictionary + runtime

**Files:**

**Interfaces:**
- Produces: globals `I18N`, `LOCALE`, `LANG`, `t(key, vars?)`, `setLang(code)`, `applyLang()`, `detectLang()`; `fmt()` locale-aware.

- [ ] **Step 1: Runtime**

```js
const LOCALE = { en:'en-US', fr:'fr-FR', de:'de-DE', ar:'ar-MA' };
let LANG = 'en';
const fmt = (n,d=0) => (isFinite(n)?n:0).toLocaleString(LOCALE[LANG], {minimumFractionDigits:d, maximumFractionDigits:d, numberingSystem:'latn'});
const t = (k, v) => { let s = (I18N[LANG] && I18N[LANG][k]) ?? I18N.en[k] ?? k; if (v) for (const p in v) s = s.split('{'+p+'}').join(v[p]); return s; };
function applyLang(){
  document.documentElement.lang = LANG; document.documentElement.dir = LANG==='ar' ? 'rtl' : 'ltr';
  document.querySelectorAll('[data-i18n]').forEach(e => e.textContent = t(e.dataset.i18n));
  document.querySelectorAll('[data-i18n-html]').forEach(e => e.innerHTML = t(e.dataset.i18nHtml));
  document.querySelectorAll('[data-i18n-attr]').forEach(e => e.dataset.i18nAttr.split(',').forEach(p => { const [a,k] = p.split(':'); e.setAttribute(a, t(k)); }));
  document.querySelectorAll('.lang-btn').forEach(b => b.setAttribute('aria-pressed', String(b.dataset.lang===LANG)));
}
function setLang(code){ if(!I18N[code]) code='en'; LANG=code; try{localStorage.setItem('zeoplant.lang',code);}catch(e){} applyLang(); calc(); }
function detectLang(){ try{ const s=localStorage.getItem('zeoplant.lang'); if(I18N[s]) return s; }catch(e){} for(const l of (navigator.languages||[navigator.language||'en'])){ const p=String(l).slice(0,2).toLowerCase(); if(I18N[p]) return p; } return 'en'; }
document.addEventListener('click', e => { const b = e.target.closest('.lang-btn'); if (b) setLang(b.dataset.lang); });
```
  Boot order: `loadInputs(); LANG = detectLang(); applyLang(); calc();`
